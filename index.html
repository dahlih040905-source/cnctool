Supabase SQL 建表語法 (建表一次即可)：
create table if not exists cnc_storage ( id text primary key default 'default_workshop', master_tools jsonb not null default '[]'::jsonb, hierarchy jsonb not null default '[]'::jsonb, custom_fields jsonb not null default '[]'::jsonb, updated_at timestamptz default now() ); alter table cnc_storage enable row level security; drop policy if exists "public_all" on cnc_storage; create policy "public_all" on cnc_storage for all using (true) with check (true); insert into cnc_storage (id) values ('default_workshop') on conflict do nothing; alter publication supabase_realtime add table cnc_storage;
